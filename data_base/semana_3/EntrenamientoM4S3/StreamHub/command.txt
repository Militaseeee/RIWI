UTILIZAR $gt -> se utiliza para mayor que

db.media.find({ duration: { $gt: 90 } })

UTILIZAR $lt -> se utiliza para menor que

db.media.find({ duration: { $lt: 110 } })

UTILIZAR $eq -> se utiliza para igual que

db.users.find({ country: "colombia" })

UTILIZAR $in -> se utiliza para decir que es en un conjunto

db.media.find({ genres: { $in: ["mystery", "drama"] } })

UTILIZAR $and -> se utiliza para decir como una condicion logica

db.media.find({ 
  $and: [
    { duration: { $gt: 120 } },
    { genres: "mystery" }
  ]
})

UTILIZAR $or -> se utiliza para decir como una condicion logica

db.users.find({ 
  $or: [
    { country: "spain" },
    { prefergenre: "horror" }
  ]
})

UTILIZAR $regex > se utiliza para buscar un texto sin importar si esta en mayuscula o minuscula

db.comments.find({ comment: { $regex: /plot/i } })

db.comments.find({ comment: { $regex: /!$/ } })

db.comments.find({ comment: { $regex: /^amazing/i } })

  { $inc: { score: 1 } }

db.ranking.updateMany(
  { score: { $lt: 4 } },           // documentos con score menor a 4
  { 
    $set: { status: "improved" },  // agregamos o actualizamos el campo status
    $inc: { score: 1, views: 1 }   // aumentamos score y views en 1
  }
)


db.ranking.updateMany(
  { score: { $lt: 4 } },           // documentos con score menor a 4
  { 
    $set: { status: "improved" },  // agregamos o actualizamos el campo status
    $inc: { score: 1, views: 1 }   // aumentamos score y views en 1
  }
)


Busca todos los documentos donde score es menor que 4.

Les cambia o agrega el campo status con el valor "improved".

Incrementa en 1 tanto el campo score como el campo views.

Si views no existe, lo crea con valor 1.

Si score es 3, después será 4; si es 2, será 3, y así.

SE UTILIZA PARA separar generos - $unwind

db.media.aggregate([
  { $unwind: "$genres" }
]).pretty()


SE UTILIZA PARA contar cuántas películas por género - $group

db.media.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", count: { $sum: 1 } } }
]).pretty()

SE UTILIZA PARA filtrar géneros con más de 1 película - $match

db.media.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", count: { $sum: 1 } } },
  { $match: { count: { $gt: 1 } } }
]).pretty()

SE UTILIZA PARA cambia nombres y añade campo nuevo - $project

db.media.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", count: { $sum: 1 } } },
  { $project: {
      genre: "$_id",
      count: 1,
      popular: { $gt: ["$count", 2] },
      _id: 0
    }
  }
]).pretty()

SE UTILIZA PARA ordenar por cantidad (de mayor a menor)	- $sort

db.media.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", count: { $sum: 1 } } },
  { $project: {
      genre: "$_id",
      count: 1,
      popular: { $gt: ["$count", 2] },
      _id: 0
    }
  },
  { $sort: { count: -1 } }
]).pretty()


SE UTILIZA PARA mostrar solo los top 3 - $limit

db.media.aggregate([
  { $unwind: "$genres" },
  { $group: { _id: "$genres", count: { $sum: 1 } } },
  { $project: {
      genre: "$_id",
      count: 1,
      popular: { $gt: ["$count", 2] },
      _id: 0
    }
  },
  { $sort: { count: -1 } },
  { $limit: 3 }
]).pretty()
