public RepairOrder assignTech(Long orderId, Long techId, Long adminActorId) {
    // 1. Verificar que quien actúa es ADMIN
    User admin = getUserById(adminActorId);
    if (admin.getRole() != Role.ADMIN) {
        throw new ForbiddenAccessException("Solo un ADMIN puede asignar técnicos.");
    }

    // 2. Encontrar la orden
    RepairOrder order = orderRepository.findById(orderId)
            .orElseThrow(() -> new ResourceNotFoundException("Order not found: "D + orderId));

    // 3. Encontrar al técnico
    User tech = getUserById(techId);

    // 4. Verificar que el usuario a asignar SÍ es un técnico
    if (tech.getRole() != Role.TECH) {
        throw new BadRequestException("El usuario " + tech.getUsername() + " no es un técnico (TECH)");
    }

    // 5. Asignar y guardar

    return orderRepository.save(order);
